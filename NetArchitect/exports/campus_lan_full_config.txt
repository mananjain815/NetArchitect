!======================================================================
! FILE: CAMPUS-L3SW.cfg
!======================================================================
enable
configure terminal
hostname CAMPUS-L3SW
!
! NetArchitect - CAMPUS multilayer switch: SVIs, RSTP, DHCP, ACLs, OSPF
! Replace every CHANGE_ME_BEFORE_DEPLOY value before production use.
!
no ip domain-lookup
service password-encryption
ip domain-name netarchitect.local
enable secret CHANGE_ME_BEFORE_DEPLOY
username netadmin privilege 15 secret CHANGE_ME_BEFORE_DEPLOY
crypto key generate rsa modulus 2048
ip ssh version 2
!
ip routing
!
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,60 priority 4096
!
vlan 999
 name NATIVE_UNUSED
vlan 10
 name STUDENTS
vlan 20
 name FACULTY
vlan 30
 name STAFF_ADMIN
vlan 40
 name IOT_AP
vlan 50
 name SERVERS
vlan 60
 name MGMT
!
ip dhcp excluded-address 10.10.0.1 10.10.0.10
ip dhcp pool STUDENTS
 network 10.10.0.0 255.255.252.0
 default-router 10.10.0.1
!
ip dhcp excluded-address 10.10.4.1 10.10.4.10
ip dhcp pool FACULTY
 network 10.10.4.0 255.255.254.0
 default-router 10.10.4.1
!
ip dhcp excluded-address 10.10.7.1 10.10.7.10
ip dhcp pool STAFF_ADMIN
 network 10.10.7.0 255.255.255.0
 default-router 10.10.7.1
!
ip dhcp excluded-address 10.10.6.1 10.10.6.10
ip dhcp pool IOT_AP
 network 10.10.6.0 255.255.255.0
 default-router 10.10.6.1
!
interface Vlan10
 description GW-STUDENTS
 ip address 10.10.0.1 255.255.252.0
 ip access-group ACL-STUDENTS-IN in
 no shutdown
!
interface Vlan20
 description GW-FACULTY
 ip address 10.10.4.1 255.255.254.0
 ip access-group ACL-FACULTY-IN in
 no shutdown
!
interface Vlan30
 description GW-STAFF_ADMIN
 ip address 10.10.7.1 255.255.255.0
 ip access-group ACL-STAFF_ADMIN-IN in
 no shutdown
!
interface Vlan40
 description GW-IOT_AP
 ip address 10.10.6.1 255.255.255.0
 ip access-group ACL-IOT_AP-IN in
 no shutdown
!
interface Vlan50
 description GW-SERVERS
 ip address 10.10.8.1 255.255.255.128
 ip access-group ACL-SERVERS-IN in
 no shutdown
!
interface Vlan60
 description GW-MGMT
 ip address 10.10.8.129 255.255.255.192
 ip access-group ACL-MGMT-IN in
 no shutdown
!
interface GigabitEthernet0/1
 description UPLINK-TO-CAMPUS-RTR
 no switchport
 ip address 10.10.16.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
interface GigabitEthernet0/2
 description TRUNK-TO-CAMPUS-ACCESS-SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
!
interface GigabitEthernet0/3
 description TRUNK-TO-CAMPUS-ACCESS-SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
!
interface Port-channel1
 description LACP-TRUNK-TO-ACCESS
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,60
!
! --- ACL definitions (applied inbound on each SVI above) ---
ip access-list extended ACL-STUDENTS-IN
 remark NetArchitect policy: role=user subnet=10.10.0.0/22
 permit udp host 0.0.0.0 host 255.255.255.255 eq 67
 permit tcp 10.10.0.0 0.0.3.255 10.10.8.0 0.0.0.127 eq 80
 permit tcp 10.10.0.0 0.0.3.255 10.10.8.0 0.0.0.127 eq 443
 permit udp 10.10.0.0 0.0.3.255 10.10.8.0 0.0.0.127 eq 53
 permit icmp 10.10.0.0 0.0.3.255 10.10.8.0 0.0.0.127 echo
 deny ip 10.10.0.0 0.0.3.255 10.10.7.0 0.0.0.255
 deny ip 10.10.0.0 0.0.3.255 10.10.8.0 0.0.0.127
 deny ip 10.10.0.0 0.0.3.255 10.10.8.128 0.0.0.63
 permit ip 10.10.0.0 0.0.3.255 any
 deny ip any any
ip access-list extended ACL-FACULTY-IN
 remark NetArchitect policy: role=user subnet=10.10.4.0/23
 permit udp host 0.0.0.0 host 255.255.255.255 eq 67
 permit tcp 10.10.4.0 0.0.1.255 10.10.8.0 0.0.0.127 eq 80
 permit tcp 10.10.4.0 0.0.1.255 10.10.8.0 0.0.0.127 eq 443
 permit udp 10.10.4.0 0.0.1.255 10.10.8.0 0.0.0.127 eq 53
 permit icmp 10.10.4.0 0.0.1.255 10.10.8.0 0.0.0.127 echo
 deny ip 10.10.4.0 0.0.1.255 10.10.7.0 0.0.0.255
 deny ip 10.10.4.0 0.0.1.255 10.10.8.0 0.0.0.127
 deny ip 10.10.4.0 0.0.1.255 10.10.8.128 0.0.0.63
 permit ip 10.10.4.0 0.0.1.255 any
 deny ip any any
ip access-list extended ACL-STAFF_ADMIN-IN
 remark NetArchitect policy: role=admin subnet=10.10.7.0/24
 permit udp host 0.0.0.0 host 255.255.255.255 eq 67
 permit ip 10.10.7.0 0.0.0.255 any
 deny ip any any
ip access-list extended ACL-IOT_AP-IN
 remark NetArchitect policy: role=iot subnet=10.10.6.0/24
 permit udp host 0.0.0.0 host 255.255.255.255 eq 67
 permit udp 10.10.6.0 0.0.0.255 host 10.10.6.1 eq 67
 permit udp 10.10.6.0 0.0.0.255 10.10.8.0 0.0.0.127 eq 53
 permit tcp 10.10.6.0 0.0.0.255 10.10.8.0 0.0.0.127 eq 443
 permit udp 10.10.6.0 0.0.0.255 10.10.8.0 0.0.0.127 eq 123
 permit tcp 10.10.6.0 0.0.0.255 10.10.8.0 0.0.0.127 eq 8883
 deny ip 10.10.6.0 0.0.0.255 10.10.0.0 0.0.255.255
 permit ip 10.10.6.0 0.0.0.255 any
 deny ip any any
ip access-list extended ACL-SERVERS-IN
 remark NetArchitect policy: role=server subnet=10.10.8.0/25
 permit tcp 10.10.8.0 0.0.0.127 any established
 permit udp 10.10.8.0 0.0.0.127 eq 53 any
 permit icmp 10.10.8.0 0.0.0.127 any echo-reply
 deny ip any any
ip access-list extended ACL-MGMT-IN
 remark NetArchitect policy: role=mgmt subnet=10.10.8.128/26
 permit ip 10.10.8.128 0.0.0.63 any
 deny ip any any
!
router ospf 1
 router-id 10.255.1.1
 passive-interface default
 no passive-interface GigabitEthernet0/1
 auto-cost reference-bandwidth 10000
 network 10.10.0.0 0.0.3.255 area 0
 network 10.10.4.0 0.0.1.255 area 0
 network 10.10.7.0 0.0.0.255 area 0
 network 10.10.6.0 0.0.0.255 area 0
 network 10.10.8.0 0.0.0.127 area 0
 network 10.10.8.128 0.0.0.63 area 0
 network 10.10.16.2 0.0.0.0 area 0
!
line console 0
 logging synchronous
 exec-timeout 10 0
!
line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
!
end
write memory
