!======================================================================
! FILE: CITY-1-L3SW.cfg
!======================================================================
enable
configure terminal
hostname CITY-1-L3SW
!
! NetArchitect - CITY_1 multilayer switch: SVIs, RSTP, DHCP, ACLs, OSPF
! Replace every CHANGE_ME_BEFORE_DEPLOY value before production use.
!
no ip domain-lookup
service password-encryption
ip domain-name netarchitect.local
enable secret CHANGE_ME_BEFORE_DEPLOY
username netadmin privilege 15 secret CHANGE_ME_BEFORE_DEPLOY
crypto key generate rsa modulus 2048
ip ssh version 2
!
ip routing
!
spanning-tree mode rapid-pvst
spanning-tree vlan 10 priority 4096
!
vlan 999
 name NATIVE_UNUSED
vlan 10
 name CITY_1_IOT
!
ip dhcp excluded-address 10.30.0.1 10.30.0.10
ip dhcp pool CITY_1_IOT
 network 10.30.0.0 255.255.254.0
 default-router 10.30.0.1
!
interface Vlan10
 description GW-CITY_1_IOT
 ip address 10.30.0.1 255.255.254.0
 ip access-group ACL-CITY_1_IOT-IN in
 no shutdown
!
interface GigabitEthernet0/1
 description UPLINK-TO-CITY-1-RTR
 no switchport
 ip address 10.30.8.130 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
interface GigabitEthernet0/2
 description TRUNK-TO-CITY-1-ACCESS-SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
!
interface GigabitEthernet0/3
 description TRUNK-TO-CITY-1-ACCESS-SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
!
interface Port-channel1
 description LACP-TRUNK-TO-ACCESS
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10
!
! --- ACL definitions (applied inbound on each SVI above) ---
ip access-list extended ACL-CITY_1_IOT-IN
 remark NetArchitect policy: role=iot subnet=10.30.0.0/23
 permit udp host 0.0.0.0 host 255.255.255.255 eq 67
 permit udp 10.30.0.0 0.0.1.255 host 10.30.0.1 eq 67
 permit udp 10.30.0.0 0.0.1.255 10.30.8.0 0.0.0.127 eq 53
 permit tcp 10.30.0.0 0.0.1.255 10.30.8.0 0.0.0.127 eq 443
 permit udp 10.30.0.0 0.0.1.255 10.30.8.0 0.0.0.127 eq 123
 permit tcp 10.30.0.0 0.0.1.255 10.30.8.0 0.0.0.127 eq 8883
 deny ip 10.30.0.0 0.0.1.255 10.30.0.0 0.0.255.255
 permit ip 10.30.0.0 0.0.1.255 any
 deny ip any any
!
router ospf 1
 router-id 10.255.1.1
 passive-interface default
 no passive-interface GigabitEthernet0/1
 auto-cost reference-bandwidth 10000
 network 10.30.0.0 0.0.1.255 area 0
 network 10.30.8.130 0.0.0.0 area 0
!
line console 0
 logging synchronous
 exec-timeout 10 0
!
line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
!
end
write memory
