!======================================================================
! FILE: HQ-L3SW.cfg
!======================================================================
enable
configure terminal
hostname HQ-L3SW
!
! NetArchitect - HQ multilayer switch: SVIs, RSTP, DHCP, ACLs, OSPF
! Replace every CHANGE_ME_BEFORE_DEPLOY value before production use.
!
no ip domain-lookup
service password-encryption
ip domain-name netarchitect.local
enable secret CHANGE_ME_BEFORE_DEPLOY
username netadmin privilege 15 secret CHANGE_ME_BEFORE_DEPLOY
crypto key generate rsa modulus 2048
ip ssh version 2
!
ip routing
!
spanning-tree mode rapid-pvst
spanning-tree vlan 10,40,50 priority 4096
!
vlan 999
 name NATIVE_UNUSED
vlan 10
 name HQ_USERS
vlan 40
 name FINTECH_SERVERS
vlan 50
 name HQ_MGMT
!
ip dhcp excluded-address 10.20.0.1 10.20.0.10
ip dhcp pool HQ_USERS
 network 10.20.0.0 255.255.255.0
 default-router 10.20.0.1
!
interface Vlan10
 description GW-HQ_USERS
 ip address 10.20.0.1 255.255.255.0
 ip access-group ACL-HQ_USERS-IN in
 no shutdown
!
interface Vlan40
 description GW-FINTECH_SERVERS
 ip address 10.20.1.1 255.255.255.192
 ip access-group ACL-FINTECH_SERVERS-IN in
 no shutdown
!
interface Vlan50
 description GW-HQ_MGMT
 ip address 10.20.1.65 255.255.255.224
 ip access-group ACL-HQ_MGMT-IN in
 no shutdown
!
interface GigabitEthernet0/1
 description UPLINK-TO-HQ-RTR
 no switchport
 ip address 10.20.3.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
interface GigabitEthernet0/2
 description TRUNK-TO-HQ-ACCESS-SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
!
interface GigabitEthernet0/3
 description TRUNK-TO-HQ-ACCESS-SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 no shutdown
!
interface Port-channel1
 description LACP-TRUNK-TO-ACCESS
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,50
!
! --- ACL definitions (applied inbound on each SVI above) ---
ip access-list extended ACL-HQ_USERS-IN
 remark NetArchitect policy: role=user subnet=10.20.0.0/24
 permit udp host 0.0.0.0 host 255.255.255.255 eq 67
 permit tcp 10.20.0.0 0.0.0.255 10.20.1.0 0.0.0.63 eq 80
 permit tcp 10.20.0.0 0.0.0.255 10.20.1.0 0.0.0.63 eq 443
 permit udp 10.20.0.0 0.0.0.255 10.20.1.0 0.0.0.63 eq 53
 permit icmp 10.20.0.0 0.0.0.255 10.20.1.0 0.0.0.63 echo
 deny ip 10.20.0.0 0.0.0.255 10.20.1.0 0.0.0.63
 deny ip 10.20.0.0 0.0.0.255 10.20.1.64 0.0.0.31
 permit ip 10.20.0.0 0.0.0.255 any
 deny ip any any
ip access-list extended ACL-FINTECH_SERVERS-IN
 remark NetArchitect policy: role=server subnet=10.20.1.0/26
 permit tcp 10.20.1.0 0.0.0.63 any established
 permit udp 10.20.1.0 0.0.0.63 eq 53 any
 permit icmp 10.20.1.0 0.0.0.63 any echo-reply
 deny ip any any
ip access-list extended ACL-HQ_MGMT-IN
 remark NetArchitect policy: role=mgmt subnet=10.20.1.64/27
 permit ip 10.20.1.64 0.0.0.31 any
 deny ip any any
!
router ospf 1
 router-id 10.255.1.1
 passive-interface default
 no passive-interface GigabitEthernet0/1
 auto-cost reference-bandwidth 10000
 network 10.20.0.0 0.0.0.255 area 0
 network 10.20.1.0 0.0.0.63 area 0
 network 10.20.1.64 0.0.0.31 area 0
 network 10.20.3.2 0.0.0.0 area 0
!
line console 0
 logging synchronous
 exec-timeout 10 0
!
line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
!
end
write memory
